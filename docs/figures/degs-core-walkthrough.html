0503  def json_values_equal(left, right):
0504      # Finite decoded JSON trees: each pair visits one corresponding subtree.
0505      pending = [(left, right)]
0506      while pending:
0507          left, right = pending.pop()
0508          if isinstance(left, bool) or isinstance(right, bool):
0509              if type(left) is not type(right) or left != right:
0510                  return False
0511          elif isinstance(left, list) or isinstance(right, list):
0512              if not (isinstance(left, list) and isinstance(right, list)):
0513                  return False
0514              if len(left) != len(right):
0515                  return False
0516              pending.extend(zip(left, right))
0517          elif isinstance(left, dict) or isinstance(right, dict):
0518              if not (isinstance(left, dict) and isinstance(right, dict)):
0519                  return False
0520              if left.keys() != right.keys():
0521                  return False
0522              pending.extend((left[key], right[key]) for key in left)
0523          elif left != right:
0524              return False
0525      return True

0621          errors.append((path or "input", "Value is not in the allowed set."))

0623      if isinstance(value, str) and len(value) < schema.get("minLength", 0):
0624          errors.append((path or "input", "String is shorter than the minimum length."))
0625      if isinstance(value, str) and "pattern" in schema:
0626          pattern = schema["pattern"]
0627          try:
0628              pattern_matches = isinstance(pattern, str) and re.search(pattern, value)
0629          except re.error:
0630              pattern_matches = False
0631          if not pattern_matches:
0632              errors.append((path or "input", "String does not match the required pattern."))
0633      if isinstance(value, list):
0634          if len(value) < schema.get("minItems", 0):
0635              errors.append((path or "input", "Array has too few items."))
0636          item_schema = schema.get("items")
0637          if isinstance(item_schema, dict):
0638              for index, item in enumerate(value):
0639                  errors.extend(
0640                      json_schema_errors(
0641                          item,
0642                          item_schema,
0643                          root_schema,
0644                          f"{path}[{index}]" if path else f"[{index}]",
0645                      )
0646                  )
0647      if isinstance(value, dict):
0648          properties = schema.get("properties", {})
0649          for required in schema.get("required", []):
0650              if required not in value:
0651                  errors.append(
0652                      (
0653                          child_schema_path(path, required),
0654                          "Required property is missing.",
0655                      )
0656                  )
0657          if isinstance(properties, dict):
0658              for key, child in value.items():
0659                  if key in properties:
0660                      errors.extend(
0661                          json_schema_errors(
0662                              child,
0663                              properties[key],
0664                              root_schema,
0665                              child_schema_path(path, key),
0666                          )
0667                      )
0668                  elif schema.get("additionalProperties") is False:
0669                      errors.append(
0670                          (
0671                              child_schema_path(path, key),
0672                              "Additional property is not allowed.",
0673                          )
0674                      )
0675      return errors

0683  def validate_structure(task):
0684      """Validate the complete canonical task schema with the standard library."""
0685      try:
0686          schema = json.loads(SCHEMA_PATH.read_text(encoding="utf-8"))
0687      except (OSError, json.JSONDecodeError) as error:
0688          return result(
0689              "FAIL",
0690              task,
0691              unmet=[
0692                  {
0693                      "field": "input",
0694                      "rule_id": "DEGS-REQ-001",
0695                      "message": f"Canonical task schema is unavailable: {error.__class__.__name__}.",
0696                  }
0697              ],
0698              rule_ids=["DEGS-REQ-001"],
0699          )
0700      raw_errors = json_schema_errors(task, schema, schema)
0701      unmet = []
0702      seen = set()
0703      for field, message in raw_errors:
0704          identity = (field, message)
0705          if identity in seen:
0706              continue
0707          seen.add(identity)
0708          add_unmet(unmet, field, schema_rule_for_path(field), message)
0709      rules = list(dict.fromkeys(item["rule_id"] for item in unmet))
0710      return result("FAIL" if unmet else "PASS", task, unmet=unmet, rule_ids=rules or ["DEGS-REQ-001"])

0951  def unresolved_critical_warning(task):
0952      for field in ("warnings", "unresolved_items"):
0953          values = task.get(field)
0954          if not isinstance(values, list):
0955              continue
0956          for item in values:
0957              if not isinstance(item, dict):
0958                  continue
0959              severity = str(item.get("severity", "")).upper()
0960              resolved = item.get("resolved") is True or str(item.get("status", "")).upper() == "RESOLVED"
0961              if severity == "CRITICAL" and not resolved:
0962                  return True
0963      return False

0967      summaries = []
0968      values = task.get("warnings")
0969      if not isinstance(values, list):
0970          return summaries
0971      for item in values:
0972          if not isinstance(item, dict):
0973              continue
0974          resolved = item.get("resolved") is True or str(item.get("status", "")).upper() == "RESOLVED"
0975          if resolved:
0976              continue
0977          severity = str(item.get("severity", "WARNING")).upper()
0978          summaries.append(
0979              {
0980                  "severity": severity,
0981                  "message": "An unresolved warning is recorded in the task input.",
0982              }

0987  def apply_warning_policy(task, payload):
0988      warnings = unresolved_warning_summaries(task)
0989      payload["warnings"] = warnings
0990      if payload["decision"] != "PASS" or not warnings:
0991          return payload
0992      if any(item["severity"] == "CRITICAL" for item in warnings):
0993          payload["decision"] = "BLOCKED"
0994          add_unmet(
0995              payload["unmet_requirements"],
0996              "warnings",
0997              "DEGS-RISK-004",
0998              "An unresolved critical warning blocks completion.",
0999          )
1000          if "DEGS-RISK-004" not in payload["applicable_rule_ids"]:
1001              payload["applicable_rule_ids"].append("DEGS-RISK-004")
1002      else:
1003          payload["decision"] = "PASS_WITH_WARNINGS"
1004      return payload

DEGS | what the core does

A deterministic evaluator for engineering task records. Current corrected source: 1,213 lines.

Figure 1
The evaluator’s main path

1. Read and shape

Parse the supplied record. Check supported schema constraints and reject invalid input.

2. Scale the controls

Choose the tier and allowed recorded authority. Higher-consequence work requires more evidence.

3. Preserve stop states

Apply warning and declared-status policies. Missing evidence and explicit failures remain visible.

4. Return a decision

Emit readiness, rule-linked diagnostics, warnings, and recorded evidence paths. Do not execute the task.

Why the boundaries matter

A source path is not proof that its contents are correct. A recorded approval is not an approval granted by the program. The code checks the structured claim against its rules; people retain responsibility for consequential evidence and action.

The staged gate includes a bounded JSON equality correction; policy rules and canonical fixture outcomes are preserved. Local activation state, private audit history, and the Workbench adapter are outside this draft. The Sashiko-derived concern collections are a separate component, not part of this gate path.

DEGS | the defect the public examples exposed

Fourteen manifestations. One comparator defect. A tested correction with its history preserved.

The distinction must survive nesting

JavaScript Object Notation (JSON) distinguishes booleans from numbers. The corrected helper preserves that distinction through arrays and objects. A const schema requiring [false] now rejects [0], while 1 and 1.0 remain equal. The first implementation delegated container equality to Python; all 14 original failures came from that defect.

What the independent review established

All 14 invalid labels and false acceptances were reproduced from the pinned public corpus: six enum cases and eight const cases. The current canonical schema has 53 const nodes and 16 enum nodes, with no container-valued comparison operands. Four scalar boolean/number substitutions failed structure checks. No current canonical task bypass was demonstrated.

What the correction preserves

An explicit worklist compares nested arrays and objects without consuming a Python call frame per level. Booleans remain distinct from numbers; 1 and 1.0 remain equal. Array order and length matter; object member order does not. Independent review caught a depth regression in the intermediate correction. New 600-level array/object tests now pass (Wright et al., 2022).

Benchmark: 241/241 matched in the unchanged supported subset. This is helper interoperability evidence, not full JSON Schema conformance or whole-system accuracy. The original 227/241 observation remains intact. Harness corrections preserve timeout diagnostics, require stale freshness, and record actual invocation arguments.
